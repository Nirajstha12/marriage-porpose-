        // RUNAWAY BUTTON MECHANISM

        const noBtn = document.getElementById('evading-button');

        const wrapper = document.getElementById('action-wrapper');


        function moveButton() {

            const wrapperBounds = wrapper.getBoundingClientRect();

            const btnBounds = noBtn.getBoundingClientRect();


            // Establish secure borders inside the main container card

            const permissibleWidth = wrapperBounds.width - btnBounds.width;

            const permissibleHeight = 160; 


            const newX = Math.floor(Math.random() * permissibleWidth);

            const newY = Math.floor((Math.random() * permissibleHeight) - (permissibleHeight / 2));


            noBtn.style.left = newX + 'px';

            noBtn.style.top = newY + 'px';

        }


        noBtn.addEventListener('mouseenter', moveButton);

        noBtn.addEventListener('touchstart', function(e) {

            e.preventDefault();

            moveButton();

        });


        // SCREEN NAVIGATION SYSTEM

        function navigate(targetStep) {

            document.querySelectorAll('.step').forEach(element => element.classList.remove('active'));

            document.getElementById('step' + targetStep).classList.add('active');

        }


        // ROW SELECTION SYSTEM

        function handleSelect(element, groupName) {

            // Uncheck other rows in the same selection group

            element.parentElement.querySelectorAll('.choice-row').forEach(row => row.classList.remove('selected'));

            

            // Mark current row as active

            element.classList.add('selected');
